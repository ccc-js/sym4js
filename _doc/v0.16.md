## 使用範例

```typescript
import { evaluate, Symbol, Exp, Pow, Sin, Cos, Gamma, Legendre, Integer } from 'sym4js'

const x = new Symbol('x')
const vars = new Map([[x, 1.5]])

// 基本運算
console.log(evaluate(new Pow(x, new Integer(2)), vars))
// 2.25

// 三角函數
console.log(evaluate(new Sin(x), vars))
// 約 0.997

// 特殊函數
console.log(evaluate(new Gamma(new Integer(5)), new Map()))
// 24 (4!)

console.log(evaluate(new Legendre(new Integer(2), x), vars))
// 約 1.4375 (對 x=1.5, P_2(1.5) = (3*2.25-1)/2 = 5.75/2 = 2.875)

console.log(evaluate(new Gamma(x), vars))
// 對 x=1.5, Gamma(1.5) ≈ 0.886
```
