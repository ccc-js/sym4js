  - 基本線性因子支援
- [x] **有理函數積分** (integrateRational)

  - `cos(2x) → cos²(x) - sin²(x)`
  - `tan(2x) → 2tan(x)/(1-tan²(x))`

## 使用範例

```typescript
import { Poly, simplify, sin, cos, tan, Symbol, Integer, Add, Div, Mul, Pow } from 'sym4js'
import { partialFractions, integrateRational } from 'sym4js'

const x = new Symbol('x')

// 有理根定理
const p = new Poly([1n, 0n, -2n, 0n])  // x³ - 2x
console.log(p.rationalRoots())  // [0n, 1n, -1n] (for x³ - 2x² - x + 2)

// 牛頓法求根
const q = new Poly([1n, 0n, -2n])  // x² - 2
const root = q.newtonRaphson(1.0)  // ≈ 1.414
console.log(root)  // 1.4142135623730951

// 部分分式
const rational = new Div(new Integer(1), new Add(new Pow(x, new Integer(2)), new Neg(new Integer(1))))
console.log(partialFractions(rational, x))
// 1/2(1/(x-1) - 1/(x+1))

// 三角倍角化簡
console.log(simplify(new Sin(new Mul(new Integer(2), x))).toString())
// 2*sin(x)*cos(x)

console.log(simplify(new Cos(new Mul(new Integer(2), x))).toString())
// cos(x)^2 - sin(x)^2
```
